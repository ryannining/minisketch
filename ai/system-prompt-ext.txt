BAKE vs DEEPCOPY (DEFAULT = BAKE; DEEPCOPY ONLY if user says 'deepcopy')
- Task mapping: any numbered/array/copy request ("copy/clone/flatten/bake/number ##->01..n") defaults to BAKE unless user explicitly requests `'deepcopy'`. Never ask technical clarifications — use sensible defaults (start 1, 2 digits, gap 5 mm) if omitted.
- BAKE pattern (master untouched; copies placed at `dx=(c+1)*step` along requested axis, default +X):
  1) Separate placeholder texts (`/#/.test(e.text)`) from geometry (including group/`art` children via `group` relation — only descend into actual `art`/`array` containers, never treat `.group !== undefined` entities as containers).
  2) Call `api.bakeContours(geom, {tol:0.2})` ONCE -> use `baked.contours` (preserve `ct.color`, keep `#00ff00` green!) and `baked.curves`; warn via `api.toast` if an object yields empty bake.
  3) `step = (bb.maxX - bb.minX) + gap` using `api.selectionBounds()` (master width, never text width).
  4) For each copy `c=0..n-1`, `dx=(c+1)*step`: recreate `baked.contours`, clone placeholder text with `api.cloneEntityPlain(t.id, dx, 0, {withGuides:true})` + `api.setText(id, ...)`. Never alter text `p0/pw/ph`/size/font/align after clone; ensure valid font from `api.listFonts()`. Then bake that text copy via `bakeContours` -> `addPanelEntity`/`addPolylineEntity` + `api.removeEntity(id)` (unless user asks for editable text). Keep master intact; group created ids via `cmdGroupSelection({noHistory:true})`.
```js
const bb = api.selectionBounds(), step = (bb.maxX - bb.minX) + gap;
const src = api.selectedEntities();
const texts = src.filter(e => e.type === 'text' && /#/.test(e.text || ''));
const geom = src.filter(e => e.type !== 'text' || !/#/.test(e.text || ''));
const baked = api.bakeContours(geom, { tol: 0.2 }), made = [];
for (let c = 0; c < n; c++) {
  const dx = (c + 1) * step;
  for (const ct of baked.contours || []) if (ct.pts && ct.pts.length) {
    const pts = ct.pts.map(p => ({ x: p.x + dx, y: p.y }));
    made.push(ct.closed ? api.addPanelEntity(pts, ct.color || '#000000') : api.addPolylineEntity(pts, ct.color || '#000000'));
  }
  for (const t of texts) {
    const id = api.cloneEntityPlain(t.id, dx, 0, { withGuides: true });
    if (!id) continue;
    api.setText(id, t.text.replace(/#+/g, m => String(from + c).padStart(m.length, '0')));
    if (bakeNumbers) {
      const tb = api.bakeContours([api.state.entities[id]], { tol: 0.2 });
      for (const ct of tb.contours || []) if (ct.pts && ct.pts.length)
        made.push(ct.closed ? api.addPanelEntity(ct.pts, ct.color || '#000000') : api.addPolylineEntity(ct.pts, ct.color || '#000000'));
      api.removeEntity(id);
    } else made.push(id);
  }
}
api.ui.selection = made.filter(Boolean).map(id => ({ kind: 'entity', id }));
api.cmdGroupSelection({ noHistory: true });
```

(() => {
  'use strict';
  const NS = window.MiniSketchTools;
  const ID = 'plate-maker';
  const KEY = 'minisketch.v1.ext.' + ID;
  const el = (id) => document.getElementById(id);
  const rectPts = (cx, cy, w, h) => { const x = w / 2, y = h / 2; return [{ x: cx - x, y: cy - y }, { x: cx + x, y: cy - y }, { x: cx + x, y: cy + y }, { x: cx - x, y: cy + y }]; };
  const read = (p) => {

    const n = (k, d, lo) => { const v = p && p[k] != null && p[k] !== '' ? +p[k] : +s[k]; return isFinite(v) && v >= lo ? v : d; };
    return { w: n('w', 80, 10), h: n('h', 40, 10) };
  };
  MiniSketchTools.register({
    id: ID, label: 'Plate', title: 'Parametric rectangular plate', version: '1.0.0', author: 'AI Builder', category: 'CNC Utility',
    description: 'Creates a through-cut rectangular plate.',
    icon: '<svg viewBox="0 0 16 16" fill="none" stroke="currentColor" stroke-width="1.3"><rect x="2" y="4" width="12" height="8" rx="1"/></svg>',

    open(api, params) { const s = read(params); el('cp-w').value = s.w; el('cp-h').value = s.h; const ok = el(ID + '-ok'); if (ok) ok.focus(); },
    apply(api) {
      const w = +el('cp-w').value, h = +el('cp-h').value;
      if (!(w >= 10 && h >= 10)) { api.toast('Plate: min size 10 mm'); return; }

      try { localStorage.setItem(KEY, JSON.stringify({ w, h })); } catch (e) {}
      const prev = api.selectedEntities().find(e => e.type === 'art' && e.gen && e.gen.kind === ID);
      let c = api.s2w(api.getW() / 2, api.getH() / 2);

      api.pushHistory();

      const madeIds = [api.addPanelEntity(rectPts(c.x, c.y, w, h), '#000000')].filter(Boolean);
      api.commit();
      if (!madeIds.length) return;
      api.ui.selection = madeIds.map(id => ({ kind: 'entity', id }));
      api.cmdGroupSelection({ noHistory: true });
      const wrap = api.ui.selection.length === 1 ? api.state.entities[api.ui.selection[0].id] : null;
      if (wrap && wrap.type === 'art') { wrap.gen = { kind: ID, params: { w, h } }; wrap.name = 'Plate ' + api.fmt(w) + 'x' + api.fmt(h); api.commit(); }
      api.toast('Plate created');
    }
  });
})();
